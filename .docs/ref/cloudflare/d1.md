# Cloudflare D1

Source: https://developers.cloudflare.com/workers/runtime-apis/bindings/d1

## wrangler.toml Configuration

```toml
[[d1_databases]]
binding = "D1"
database_name = "my-db"
database_id = "xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx"
```

## Basic Queries

```typescript
interface Env {
	DB: D1Database;
}

export default {
	async fetch(request: Request, env: Env) {
		// Run query - returns full D1Result object
		const result = await env.DB.prepare("SELECT * FROM Customers WHERE CompanyName = ?").bind("Bs Beverages").run();

		// Raw - returns array of arrays
		const raw = await env.DB.prepare("SELECT * FROM users").raw();

		// First - returns first row only
		const first = await env.DB.prepare("SELECT * FROM users WHERE id = ?").bind(userId).first();

		// Batch - execute multiple statements
		const batchResult = await env.DB.batch([
			env.DB.prepare("SELECT * FROM users"),
			env.DB.prepare("SELECT * FROM orders"),
		]);

		return Response.json(result);
	},
} satisfies ExportedHandler<Env>;
```

## Typed Queries

```typescript
type OrderRow = {
	Id: string;
	CustomerName: string;
	OrderDate: number;
};

const result = await env.DB.prepare(
	"SELECT Id, CustomerName, OrderDate FROM [Order] ORDER BY ShippedDate DESC LIMIT 100"
).run<OrderRow>();
```

## D1 Sessions (Read Replication)

```typescript
const session = env.DB.withSession("primary-reader");
const result = await session.prepare("SELECT * FROM users").run();
```

## Miniflare Testing

```typescript
const db = await mf.getD1Database("DB");
const stmt = await db.prepare("<Query>");
const result = await stmt.run();

return Response.json(result.results);
```

## Python Worker

```python
from workers import WorkerEntrypoint, Response

class Default(WorkerEntrypoint):
    async def fetch(self, request):
        results = await self.env.DB.prepare("PRAGMA table_list").run()
        return Response.json(results)
```
